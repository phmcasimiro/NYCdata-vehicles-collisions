-- 1. NOVA VIEW TEMPORAL E REGIONAL (Baseada em crash_timestamp)
CREATE MATERIALIZED VIEW public.nycdata_vehicles_collisions_gold_fact_temporal AS
SELECT 
    EXTRACT(YEAR FROM DATE_TRUNC('month', crash_timestamp))::INTEGER as year,
    DATE_TRUNC('month', crash_timestamp)::DATE as period_date,
    COALESCE(NULLIF(borough, ''), 'UNKNOWN') as borough,
    COUNT(*) as total_collisions,
    SUM(COALESCE(total_persons_killed, 0)) as total_fatalities,
    SUM(COALESCE(total_persons_injured, 0) + COALESCE(total_persons_killed, 0)) as people_involved
FROM public.nycdata_vehicle_collisions_cleaned
GROUP BY DATE_TRUNC('month', crash_timestamp), borough;

-- Índices de Performance e Unicidade Exigidos pelo pg_cron
CREATE UNIQUE INDEX idx_gold_temporal_unique ON public.nycdata_vehicles_collisions_gold_fact_temporal (period_date, borough);


-- 2. NOVA VIEW DE FATORES CONTRIBUINTES (Baseada na coluna canônica do seu banco)
CREATE MATERIALIZED VIEW public.nycdata_vehicles_collisions_gold_fact_contributing_factors AS
SELECT 
    COALESCE(NULLIF(borough, ''), 'UNKNOWN') as borough,
    contributing_factor,
    COUNT(*) as total_collisions
FROM (
    SELECT borough, contributing_factor_vehicle_1 as contributing_factor FROM public.nycdata_vehicle_collisions_cleaned
) as unified_factors
WHERE contributing_factor IS NOT NULL 
  AND contributing_factor NOT IN ('', 'UNSPECIFIED', 'UNKNOWN')
GROUP BY borough, contributing_factor;

CREATE INDEX idx_gold_factors_borough ON public.nycdata_vehicles_collisions_gold_fact_contributing_factors (borough);
